244FIFTYONE_DEGREES_RESULTS_INLINE void fiftyoneDegreesResultsEvaluate(
245 fiftyoneDegreesResultsBase *results,
246 byte componentIndex,
247 fiftyoneDegreesException *exception) {
248 uint32_t *word = &results->pending[componentIndex >> 5];
249 const uint32_t bit = (uint32_t)1 << (componentIndex & 31);
250 if ((*word & bit) != 0) {
251 *word &= ~bit;
252 if (results->evaluate != NULL) {
253 results->evaluate(results, componentIndex, exception);
254 }
255 }
256}

267FIFTYONE_DEGREES_RESULTS_INLINE void fiftyoneDegreesResultsEvaluateAll(
268 fiftyoneDegreesResultsBase *results,
269 fiftyoneDegreesException *exception) {
270 uint32_t word, bit;
271 for (word = 0; word < FIFTYONE_DEGREES_RESULTS_PENDING_WORDS; word++) {
272 while (results->pending[word] != 0) {
273 bit = 0;
274 while ((results->pending[word] & ((uint32_t)1 << bit)) == 0) {
275 bit++;
276 }
277 fiftyoneDegreesResultsEvaluate(
278 results,
279 (byte)(word * 32 + bit),
280 exception);
281 if (FIFTYONE_DEGREES_EXCEPTION_FAILED) {
282 return;
283 }
284 }
285 }
286}
