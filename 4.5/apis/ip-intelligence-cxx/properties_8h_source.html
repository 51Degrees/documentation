403FIFTYONE_DEGREES_PROPERTIES_INLINE int fiftyoneDegreesPropertiesGetAvailableIndex(
404 const fiftyoneDegreesPropertiesAvailable *available,
405 int requiredPropertyIndex) {
406 if (available == NULL || requiredPropertyIndex < 0) {
407 return -1;
408 }
409 if (available->named != NULL) {
410 return (uint32_t)requiredPropertyIndex < available->namedCount ?
411 (int)available->named[requiredPropertyIndex].availableIndex :
412 -1;
413 }
414 return (uint32_t)requiredPropertyIndex < available->count ?
415 requiredPropertyIndex :
416 -1;
417}
