65 static string getString(
66 fiftyoneDegreesCollection *stringsCollection,
67 uint32_t offset) {
68 FIFTYONE_DEGREES_EXCEPTION_CREATE;
69 string result;
70 fiftyoneDegreesCollectionItem item;
71 fiftyoneDegreesDataReset(&item.data);
72 const fiftyoneDegreesStoredBinaryValue * const value =
73 fiftyoneDegreesStoredBinaryValueGet(
74 stringsCollection,
75 offset,
76 FIFTYONE_DEGREES_PROPERTY_VALUE_TYPE_STRING,
77 &item,
78 exception);
79 FIFTYONE_DEGREES_EXCEPTION_THROW;
80 if (value != nullptr) {
81 char buffer[FIFTYONE_DEGREES_REASONABLE_WKT_STRING_LENGTH];
82 // A string has no decimal places.
83 char * const text = fiftyoneDegreesStoredBinaryValueToText(
84 value,
85 FIFTYONE_DEGREES_PROPERTY_VALUE_TYPE_STRING,
86 0,
87 buffer,
88 sizeof(buffer),
89 exception);
90 FIFTYONE_DEGREES_COLLECTION_RELEASE(
91 stringsCollection,
92 &item);
93 result = takeText(text, buffer, exception);
94 }
95 return result;
96 }

107 static string takeText(
108 char *text,
109 const char *buffer,
110 fiftyoneDegreesException *exception) {
111 string result;
112 if (text != nullptr) {
113 result = text;
114 if (text != buffer) {
115 fiftyoneDegreesFree(text);
116 }
117 }
118 // Not used when exceptions are disabled.
119 (void)exception;
120 FIFTYONE_DEGREES_EXCEPTION_THROW;
121 return result;
122 }

156 case FIFTYONE_DEGREES_VALUE_TEXT_URL:

159 }
160 }

162 texts->failed = true;
163 }
