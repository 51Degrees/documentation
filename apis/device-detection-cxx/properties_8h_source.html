376 fiftyoneDegreesPropertiesGetRequiredPropertyCount(
377 const fiftyoneDegreesPropertiesAvailable *available) {
378 if (available == NULL) {
379 return 0;

382}

392FIFTYONE_DEGREES_PROPERTIES_INLINE int fiftyoneDegreesPropertiesGetAvailableIndex(
393 const fiftyoneDegreesPropertiesAvailable *available,
394 int requiredPropertyIndex) {
395 if (available == NULL || requiredPropertyIndex < 0) {
396 return -1;
397 }
398 if (available->named != NULL) {
399 return (uint32_t)requiredPropertyIndex < available->namedCount ?
400 (int)available->named[requiredPropertyIndex].availableIndex :
401 -1;
402 }
403 return (uint32_t)requiredPropertyIndex < available->count ?
404 requiredPropertyIndex :
405 -1;
406}

434 fiftyoneDegreesPropertiesGetPropertyIndexFromAvailableIndex(
435 const fiftyoneDegreesPropertiesAvailable *available,
436 int availableIndex) {
437 if (available == NULL ||
438 availableIndex < 0 ||
439 (uint32_t)availableIndex >= available->count) {
440 return -1;
