247 static unsigned long getHashCode(
248 DeviceDetection::Hash::ResultsHash* results) {
249 unsigned long hashCode = 0;
250 Common::Value<string> value;
251
252 for (const string &name : results->getProperties()) {
253 value = results->getValueAsString(name);
254 if (value.hasValue()) {
255 hashCode ^= generateHash(
256 (unsigned char*)(value.getValue().c_str()));
257 }
258 }
259 return hashCode;
260 }
