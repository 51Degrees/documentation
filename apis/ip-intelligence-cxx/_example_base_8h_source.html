47// Times in milliseconds. Windows uses the performance counter, as the tick
48// count only moves every 15 or 16 milliseconds, which is as long as some
49// timed runs take.
50#ifdef _MSC_VER
51#define TIMER_CREATE LARGE_INTEGER start, end, frequency
52#define TIMER_START (QueryPerformanceFrequency(&frequency), \
53QueryPerformanceCounter(&start))
54#define TIMER_END QueryPerformanceCounter(&end)
55#define TIMER_ELAPSED ((double)(end.QuadPart - start.QuadPart) * 1000.0 / \
56(double)frequency.QuadPart)
57#else
58#define TIMER_CREATE struct timespec start, end
59#define TIMER_START clock_gettime(CLOCK_MONOTONIC, &start)
60#define TIMER_END clock_gettime(CLOCK_MONOTONIC, &end)
61#define TIMER_ELAPSED (((end.tv_sec - start.tv_sec) + \
62(end.tv_nsec - start.tv_nsec) / 1.0e9) * 1000.0)
63#endif
